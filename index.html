    Bagaimana ini bekerja (kode SDK asli)

    Setiap langkah di halaman ini memanggil fungsi SDK yang sama seperti di bawah. Bukan simulasi — ini yang benar-benar dijalankan di browser kamu.

    // 1. Verifikasi core + baca template (sekali, di blok final)
const head = await resolveAmmoraFinalizedHead([publicClient], giwaSepoliaAutoRoute.finality);
await verifyAmmoraV4CoreAtBlock(publicClient, head);
const template = await getAmmoraV4ConfigAtBlock(publicClient, head, TEMPLATE_ID); // 6

// 2. Quick Create — derive Config dari template, lalu satu transaksi:
//    Config + token + kurva (to = factory, data + value dari builder)
const template = await getAmmoraV4ConfigAtBlock(publicClient, REGISTRY, 6n, head);
const policy = parseTeamRecipientPolicy({ schemaVersion: 1, chainId: 91342, registry: REGISTRY,
  templateId: "6", templateHash: template.configHash,
  configOwner: account, partnerRecipient: account, liquidityPartner: account, leftoverReceiver: account });
const team = planTeamConfig(template, policy, { chainId: 91342, registry: REGISTRY }, saleSupplyRaw, quoteThresholdRaw);
const built = buildRegisterConfigAndCreateAmmoraV4Launch(team.config, team.curve, {
  name, symbol, metadataURI, creatorMetadataURI: metadataURI, tokenDecimals: team.config.tokenDecimals,
  callback: zeroAddress, callbackData: "0x",
});
await walletClient.sendTransaction({ to: FACTORY, data: built.data, value: built.value });
const receipt = await publicClient.waitForTransactionReceipt({ hash, confirmations: 13 });
const config = parseAmmoraV4ConfigCreatedReceipt(receipt, REGISTRY);
const { token, curve } = parseAmmoraV4LaunchCreatedReceipt(receipt, FACTORY); // dibaca dari event, bukan prediksi

// 3. Atomic Create + First Buy — Config + token + kurva + first buy (satu tx)
const { plan } = await planTeamAtomicFromCatalog(publicClient, head, policy, {
  account, saleSupplyRaw, quoteThresholdRaw, name, symbol, metadataURI,
  quoteRaw, minimumBaseRaw: 1n, maxFeePerGasWei: gasPrice * 2n, deadline,
});
if (plan.status === "APPROVAL_REQUIRED") { await walletClient.sendTransaction(plan.transaction); /* lalu re-plan di blok baru */ }
if (plan.status === "READY") { await walletClient.sendTransaction(plan.transaction); }
const swap = parseAmmoraV4SwapReceipt(receipt, launch.curve); // trader harus = first-buy executor

// 4. Trade di kurva — quote dulu, klasifikasi fill, baru kirim
const sdk = new AmmoraV4LaunchCurveClient(publicClient, curve, { launchFactory: FACTORY, feePolicy: FEE_POLICY });
const plan = await planCurveAtBlock(publicClient, CORE, curve, { blockNumber: head, tokenIn, amountIn,
  recipient: account, deadline: block.timestamp + 1200n, slippageBps: 100 });
const fill = classifyAmmoraV4ExactInputFill(plan.quote, plan.baseForQuote); // ordinary fills harus "full"
const prepared = await PreparedExactInputSwap.create({ client: executionClient, chainId: 91342, payer: account,
  quoteTtlMs: 30_000, createPlan: async () => ({ transaction: plan.transaction,
    approvalRequirements: plan.approvalRequirements, quoteBlockNumber: head,
    quote: { model: "dbc", pool: curve, tokenIn, inputAsset: "erc20", tokenOut: plan.tokenOut, amountIn,
      fillableAmountIn: plan.quote.amountIn, remainderAmountIn: plan.quote.amountLeft,
      expectedAmountOut: plan.quote.amountOut, minimumAmountOut: plan.quote.minimumAmountOut,
      feeAmount: plan.quote.fee, feeToken: null, priceImpactBps: null, recipient: account, details: { snapshotBlock: head, snapshotHash: block.hash, policy: giwaSepoliaAutoRoute.finality, note: "Confirmed snapshot quote, not a latest execution-price guarantee.", quote: plan.quote } } }) });
const { receipt, amountReceived } = await prepared.execute(walletClient, { onStatus: (s) => console.log(s.type) });

// 5. Migrate setelah threshold tercapai
const sdk = new AmmoraV4LaunchCurveClient(publicClient, curve, { launchFactory: FACTORY, feePolicy: FEE_POLICY });
const plan = await sdk.planMigrateAtBlock(head);
await walletClient.sendTransaction({ to: plan.transaction.to, data: plan.transaction.data });

// 6. Daftar launch asli — Data API dulu, fallback scan event on-chain
const api = new AmmoraDataApiClient("https://ammora-giwa-sepolia-data.fly.dev");
const { items } = await api.getLaunches({ chainId: "91342" }); // Data API (saat ini 503)
const logs = await publicClient.getLogs({ address: FACTORY, event: parseAbiItem("event LaunchCreated(address indexed token,address indexed curve,address indexed creator,uint256 configId,bytes32 configHash,uint16 protocolVersion,string metadataURI)"), fromBlock: 36_220_000n, toBlock: head });

    Catatan: halaman ini membaca data langsung dari GIWA Sepolia RPC dan Ammora Data API. Saat Data API tidak menjawab (503), daftar launch dibaca dari event LaunchCreated on-chain.
